-- ========================================
-- 普通模式 FPGA 数据库迁移 SQL
-- ========================================
-- 数据库: fpga_static_analysis_filter
-- 说明: 普通模式 FPGA 使用独立的4张表（nm_fpga_*），与工具模式 FPGA 共用数据库
-- 创建时间: 2025.01.24
-- 
-- 表清单:
--   1. nm_fpga_analysis_tasks       - FPGA 任务表
--   2. nm_fpga_analysis_files       - FPGA 文件表
--   3. nm_fpga_analysis_functions   - FPGA 文件单元表（占位行，FPGA 按文件扫描）
--   4. nm_fpga_analysis_problems    - FPGA 问题表
-- ========================================

-- 1. 普通模式 FPGA 任务表
CREATE TABLE nm_fpga_analysis_tasks (
    id               SERIAL PRIMARY KEY,
    task_name        VARCHAR(255) NOT NULL,
    task_description TEXT,
    status           VARCHAR(50)  NOT NULL DEFAULT 'pending',  -- pending, processing, completed, failed
    progress         INTEGER      NOT NULL DEFAULT 0,          -- 0-100
    total_files      INTEGER      NOT NULL DEFAULT 0,
    processed_files  INTEGER      NOT NULL DEFAULT 0,
    total_problems   INTEGER      NOT NULL DEFAULT 0,
    code_file_id     INTEGER,                                  -- 关联的代码文件ID
    domain           VARCHAR(50)  NOT NULL DEFAULT 'normalModeFpga',  -- 固定为 normalModeFpga
    created_at       TIMESTAMP    NOT NULL DEFAULT NOW(),
    updated_at       TIMESTAMP    NOT NULL DEFAULT NOW()
);

CREATE INDEX idx_nm_fpga_tasks_status ON nm_fpga_analysis_tasks(status);
CREATE INDEX idx_nm_fpga_tasks_domain ON nm_fpga_analysis_tasks(domain);

COMMENT ON TABLE nm_fpga_analysis_tasks IS '普通模式 FPGA 任务表';
COMMENT ON COLUMN nm_fpga_analysis_tasks.domain IS '固定为 normalModeFpga，用于区分 C 模式';

-- 2. 普通模式 FPGA 文件表
CREATE TABLE nm_fpga_analysis_files (
    id              SERIAL PRIMARY KEY,
    task_id         INTEGER REFERENCES nm_fpga_analysis_tasks(id) ON DELETE CASCADE,
    file_path       VARCHAR(512) NOT NULL,  -- ZIP 文件路径
    extract_path    VARCHAR(512),           -- 解压目录路径
    file_name       VARCHAR(255) NOT NULL,
    file_size       INTEGER,                -- 文件大小(字节)
    file_type       VARCHAR(50),            -- 文件类型(zip)
    status          VARCHAR(50)  NOT NULL DEFAULT 'uploaded',  -- uploaded, processing, completed, failed
    functions_count INTEGER      NOT NULL DEFAULT 0,          -- 函数数量（FPGA 为文件数量）
    file_count      INTEGER      NOT NULL DEFAULT 0,          -- 包含的源文件数量
    upload_time     TIMESTAMP    NOT NULL DEFAULT NOW()
);

CREATE INDEX idx_nm_fpga_files_task_id ON nm_fpga_analysis_files(task_id);

COMMENT ON TABLE nm_fpga_analysis_files IS '普通模式 FPGA 文件表';
COMMENT ON COLUMN nm_fpga_analysis_files.functions_count IS 'FPGA 模式下表示 Verilog/VHDL 文件数量';

-- 3. 普通模式 FPGA 函数/文件单元表
-- 说明: FPGA 按文件扫描（不是函数），此表存储占位行，一个文件对应一条记录
CREATE TABLE nm_fpga_analysis_functions (
    id                   SERIAL PRIMARY KEY,
    file_id              INTEGER NOT NULL REFERENCES nm_fpga_analysis_files(id) ON DELETE CASCADE,
    function_name        VARCHAR(255) NOT NULL,  -- FPGA 模式下存储文件名（如 "module_name.v"）
    start_line           INTEGER NOT NULL,       -- 文件起始行（通常为 1）
    end_line             INTEGER NOT NULL,       -- 文件结束行
    source_code          TEXT    NOT NULL,       -- 文件完整源码
    analyzed             BOOLEAN NOT NULL DEFAULT FALSE,
    file_path            VARCHAR(512),           -- 在 ZIP 包中的相对路径
    function_description TEXT,                   -- AI 生成的文件功能描述（可选）
    commented_code       TEXT                    -- AI 生成的注释后代码（可选）
);

CREATE INDEX idx_nm_fpga_functions_file_id ON nm_fpga_analysis_functions(file_id);

COMMENT ON TABLE nm_fpga_analysis_functions IS '普通模式 FPGA 文件单元表（占位行，FPGA 按文件扫描）';
COMMENT ON COLUMN nm_fpga_analysis_functions.function_name IS 'FPGA 模式下存储文件名，不是真正的函数名';
COMMENT ON COLUMN nm_fpga_analysis_functions.source_code IS 'FPGA 模式下存储完整文件源码';

-- 4. 普通模式 FPGA 问题表
CREATE TABLE nm_fpga_analysis_problems (
    id                      SERIAL PRIMARY KEY,
    task_id                 INTEGER NOT NULL REFERENCES nm_fpga_analysis_tasks(id) ON DELETE CASCADE,
    file_id                 INTEGER NOT NULL REFERENCES nm_fpga_analysis_files(id) ON DELETE CASCADE,
    function_id             INTEGER NOT NULL REFERENCES nm_fpga_analysis_functions(id) ON DELETE CASCADE,  -- FPGA 模式指向文件单元
    file_name               VARCHAR(255) NOT NULL,
    source_code             TEXT    NOT NULL,       -- 问题相关的代码片段
    violation_rules         VARCHAR(255),           -- 违反的规则编号（如 "R-2-1, A-4-3"）
    issue_level             VARCHAR(50),            -- 问题等级：建议改进、一般问题、重要问题
    issue_name              VARCHAR(255),           -- 问题名称
    issue_location          VARCHAR(255),           -- 问题位置（文件名:行号）
    description_requirement TEXT,                   -- 规则要求描述
    description_issue       TEXT,                   -- 问题描述（四段式）
    description_location    TEXT,                   -- 问题定位
    description_impact      TEXT,                   -- 影响分析
    created_at              TIMESTAMP NOT NULL DEFAULT NOW()
);

CREATE INDEX idx_nm_fpga_problems_task_id  ON nm_fpga_analysis_problems(task_id);
CREATE INDEX idx_nm_fpga_problems_file_id  ON nm_fpga_analysis_problems(file_id);
CREATE INDEX idx_nm_fpga_problems_func_id  ON nm_fpga_analysis_problems(function_id);

COMMENT ON TABLE nm_fpga_analysis_problems IS '普通模式 FPGA 问题表';
COMMENT ON COLUMN nm_fpga_analysis_problems.function_id IS 'FPGA 模式下指向 nm_fpga_analysis_functions 的文件单元';
COMMENT ON COLUMN nm_fpga_analysis_problems.violation_rules IS 'FPGA 规则编号，如 R-2-1（Verilog）或 5.1.1（VHDL）';
