-- 普通模式任务表
-- 2026.7.16
CREATE TABLE nm_analysis_tasks (
    id          SERIAL PRIMARY KEY,
    task_name   VARCHAR(255) NOT NULL,
    task_description TEXT,
    status      VARCHAR(50)  NOT NULL DEFAULT 'pending',
    progress    INTEGER      NOT NULL DEFAULT 0,
    total_files INTEGER      NOT NULL DEFAULT 0,
    processed_files INTEGER  NOT NULL DEFAULT 0,
    total_problems  INTEGER  NOT NULL DEFAULT 0,
    code_file_id    INTEGER,
    domain      VARCHAR(50)  NOT NULL DEFAULT 'normalModeC',  -- 'normalModeC' 或 'normalModeFpga'
    created_at  TIMESTAMP    NOT NULL DEFAULT NOW(),
    updated_at  TIMESTAMP    NOT NULL DEFAULT NOW()
);

CREATE INDEX idx_nm_tasks_status  ON nm_analysis_tasks(status);
CREATE INDEX idx_nm_tasks_domain  ON nm_analysis_tasks(domain);

-- 普通模式文件表
CREATE TABLE nm_analysis_files (
    id           SERIAL PRIMARY KEY,
    task_id      INTEGER REFERENCES nm_analysis_tasks(id) ON DELETE CASCADE,
    file_path    VARCHAR(512) NOT NULL,
    extract_path VARCHAR(512),
    file_name    VARCHAR(255) NOT NULL,
    file_size    INTEGER,
    file_type    VARCHAR(50),
    status       VARCHAR(50)  NOT NULL DEFAULT 'uploaded',
    functions_count INTEGER   NOT NULL DEFAULT 0,
    file_count   INTEGER      NOT NULL DEFAULT 0,
    upload_time  TIMESTAMP    NOT NULL DEFAULT NOW()
);

CREATE INDEX idx_nm_files_task_id ON nm_analysis_files(task_id);

-- 普通模式问题表

CREATE TABLE nm_analysis_problems (
    id           SERIAL PRIMARY KEY,
    task_id      INTEGER NOT NULL REFERENCES nm_analysis_tasks(id) ON DELETE CASCADE,
    file_id      INTEGER NOT NULL REFERENCES nm_analysis_files(id) ON DELETE CASCADE,
    function_id  INTEGER NOT NULL DEFAULT 0,  -- 保留兼容，FPGA可存0
    file_name    VARCHAR(255) NOT NULL,
    source_code  TEXT    NOT NULL,
    violation_rules     VARCHAR(255),
    issue_level         VARCHAR(50),
    issue_name          VARCHAR(255),
    issue_location      VARCHAR(255),
    description_requirement TEXT,
    description_issue   TEXT,
    description_location TEXT,
    description_impact  TEXT,
    created_at   TIMESTAMP NOT NULL DEFAULT NOW()
);

CREATE INDEX idx_nm_problems_task_id ON nm_analysis_problems(task_id);
CREATE INDEX idx_nm_problems_file_id ON nm_analysis_problems(file_id);

-- 普通模式函数表（C模式按函数扫描，FPGA可忽略此表但需保留结构）
CREATE TABLE nm_analysis_functions (
    id                SERIAL PRIMARY KEY,
    file_id              INTEGER NOT NULL REFERENCES nm_analysis_files(id) ON DELETE CASCADE,
    function_name        VARCHAR(255) NOT NULL,
    start_line           INTEGER NOT NULL,
    end_line             INTEGER NOT NULL,
    source_code          TEXT NOT NULL,
    analyzed             BOOLEAN NOT NULL DEFAULT FALSE,
    file_path            VARCHAR(512),
    function_description TEXT,
    commented_code       TEXT
);

CREATE INDEX idx_nm_functions_file_id ON nm_analysis_functions(file_id);

ALTER TABLE nm_analysis_problems
    ADD CONSTRAINT fk_nm_problems_function
    FOREIGN KEY (function_id) REFERENCES nm_analysis_functions(id) ON DELETE CASCADE;
