-- 2026-10-12 静态分析自动修复功能（新建修复相关库表）
-- 说明：本次自动修复功能属于 C语言(filter) 模块，本段 SQL 需在 filter 所在数据库执行
--       （默认库 static_analysis_filter，用户 static_analysis）。
-- 在 Navicat 中执行以下 SQL

-- 1. 问题表补充自动修复状态列
--    fix_status: 0=未修复 1=已生成待确认 2=已应用 3=已忽略
--    fix_id    : 关联最近一次修复记录 static_analysis_fixes.id
ALTER TABLE static_analysis_issues
    ADD COLUMN IF NOT EXISTS fix_status INTEGER DEFAULT 0;
ALTER TABLE static_analysis_issues
    ADD COLUMN IF NOT EXISTS fix_id INTEGER;
CREATE INDEX IF NOT EXISTS idx_issues_fix_status ON static_analysis_issues(fix_status);

-- 2. 修复任务表（一次修复任务：单问题 / 文件级 / 规则级）
CREATE TABLE IF NOT EXISTS static_analysis_fix_tasks (
    id              SERIAL PRIMARY KEY,
    task_id         INTEGER NOT NULL,              -- 关联 static_analysis_sessions.id
    rule_type       VARCHAR(20),                   -- gjb / cwe / mixed
    scope_type      VARCHAR(20),                   -- single / file / rule
    scope_value     TEXT,                          -- problem_id / file_path / rule_code
    user_prompt     TEXT,                          -- 用户自定义修复提示词（可选）
    options_json    TEXT,
    status          VARCHAR(20) DEFAULT 'pending', -- pending/generating/done/partial/error
    total           INTEGER DEFAULT 0,
    succeeded       INTEGER DEFAULT 0,
    failed          INTEGER DEFAULT 0,
    conflicted      INTEGER DEFAULT 0,
    created_by      VARCHAR(100),
    created_at      TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at      TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
CREATE INDEX IF NOT EXISTS idx_fix_tasks_task_id ON static_analysis_fix_tasks(task_id);

-- 3. 单条修复记录表（一条 diff = 一条记录）
CREATE TABLE IF NOT EXISTS static_analysis_fixes (
    id               SERIAL PRIMARY KEY,
    fix_task_id      INTEGER NOT NULL,             -- 关联 static_analysis_fix_tasks.id
    problem_id       INTEGER,                      -- 关联 static_analysis_issues.id
    file_path        VARCHAR(512),
    line_no          INTEGER,
    rule_code        VARCHAR(50),
    rule_type        VARCHAR(20),
    original_snippet TEXT,
    fixed_snippet    TEXT,
    diff_text        TEXT,
    ai_explanation   TEXT,
    can_fix          BOOLEAN DEFAULT TRUE,
    confidence       DOUBLE PRECISION,
    status           VARCHAR(20) DEFAULT 'generated', -- generated/applied/conflict/failed/rolled_back
    error_message    TEXT,
    raw_response     TEXT,
    backup_path      VARCHAR(512),
    applied_at       TIMESTAMP,
    created_at       TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
CREATE INDEX IF NOT EXISTS idx_fixes_task_id ON static_analysis_fixes(fix_task_id);
CREATE INDEX IF NOT EXISTS idx_fixes_problem_id ON static_analysis_fixes(problem_id);

-- 4. 查看修复相关表是否创建成功
SELECT 'static_analysis_fix_tasks' AS tbl, COUNT(*) AS cnt FROM static_analysis_fix_tasks
UNION ALL
SELECT 'static_analysis_fixes' AS tbl, COUNT(*) AS cnt FROM static_analysis_fixes;
